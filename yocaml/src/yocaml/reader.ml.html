module Over (T : sig
  type env
  type 'a monad

  val return : 'a -> 'a monad
  val bind : ('a -> 'b monad) -> 'a monad -> 'b monad
end) =
struct
  type env = T.env
  type 'a monad = 'a T.monad
  type 'a t = env -> 'a monad

  let return x _ = T.return x
  let bind f reader env = T.bind (fun x -> (f x) env) (reader env)
end
