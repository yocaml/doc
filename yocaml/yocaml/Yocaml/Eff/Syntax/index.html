Sourceval (let+) : 'a t -> ('a -> 'b) -> 'b t

let+ x = e in f x is f <$> x

Sourceval (and+) : 'a t -> 'b t -> ('a * 'b) t

let+ x = e and+ y = f in g x y is g <$> e <*> f.

Sourceval (let*) : 'a t -> ('a -> 'b t) -> 'b t
